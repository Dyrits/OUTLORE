import sys; sys.path.insert(0, "/tmp/scratch-panes")
from common import *
def panel_btns(page): return page.evaluate("[...document.querySelectorAll('[aria-label=\"Show quest panel\"],[aria-label=\"Hide quest panel\"]')].map(e=>e.getAttribute('aria-label'))")
def edges(page): return page.evaluate("[...document.querySelectorAll('[aria-label=\"Resize the panes\"]')].map(e=>Math.round(e.getBoundingClientRect().x+4))")
def nth_btn(page, i): return page.locator('[aria-label="Show quest panel"],[aria-label="Hide quest panel"]').nth(i)
with sync_playwright() as p:
    page = open_page(p, W=1700)
    page.mouse.move(700, 400); page.mouse.move(2, 400); page.wait_for_timeout(500)
    page.get_by_text("Right sidebar").first.click(); page.wait_for_timeout(2000)
    page.get_by_label(re.compile("^Errand: Build the component")).first.click(); page.wait_for_timeout(1500)
    print("edges (resizers) with 3 panes:", edges(page))
    nth_btn(page,0).click(); page.wait_for_timeout(800); print("press quest icon:", panel_btns(page))
    nth_btn(page,1).click(); page.wait_for_timeout(800); print("press errand icon:", panel_btns(page))
    nth_btn(page,0).click(); page.wait_for_timeout(800); print("press quest icon again (panel was on errand):", panel_btns(page))
    nth_btn(page,0).click(); page.wait_for_timeout(800); print("press quest icon once more (same pane): ", panel_btns(page), "tabs", page.evaluate("[...document.querySelectorAll('[role=tab]')].length"))
    page.screenshot(path=f"{OUT}/d5-details-closed.png")
    # Equal widths
    print("edges:", edges(page))
    r = page.get_by_label("Resize the panes").first.bounding_box()
    page.mouse.move(r["x"]+4, 400); page.mouse.down(); page.mouse.move(r["x"]+84, 400, steps=8); page.mouse.up(); page.wait_for_timeout(500)
    print("edges after drag +80:", edges(page))
    page.get_by_label("Equal widths").first.click(); page.wait_for_timeout(600)
    print("edges after Equal widths:", edges(page))
    print("stored weights:", page.evaluate("localStorage.getItem('outlore:pane-weights-2')"))
    page.screenshot(path=f"{OUT}/d6-equal.png")
